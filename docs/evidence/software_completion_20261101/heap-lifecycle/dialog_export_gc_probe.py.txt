"""Real current plot export dialog format transition, unchanged product source."""
import gc
import json
import os
from pathlib import Path
import threading
from concurrent.futures import ThreadPoolExecutor

os.environ['QT_QPA_PLATFORM']='offscreen'
import numpy as np
from PySide6.QtCore import Qt, QEvent
from PySide6.QtWidgets import QApplication
from pyqtgraph.exporters import ImageExporter, CSVExporter
from underline_retldc.gui.analysis_widgets import AnalysisPlotWidget
from underline_retldc.i18n.service import TranslationService

out=Path(__file__).with_name('dialog-export-gc-before.jsonl')
def record(kind, **values):
    with out.open('a',encoding='utf8') as stream:
        stream.write(json.dumps(dict(kind=kind,thread=threading.get_ident(),**values))+'\n')
app=QApplication([])
plot=AnalysisPlotWidget(TranslationService(),regions_movable=False)
plot.add_series(np.arange(10.),np.arange(10.),name='probe')
plot.show()
app.processEvents()
record('main')
dialog=plot.plot_widget.scene().exportDialog
dialog.updateItemList(select=plot.plot_widget.getPlotItem())
image_row=next(i for i in range(dialog.ui.formatList.count()) if dialog.ui.formatList.item(i).expClass == ImageExporter)
csv_row=next(i for i in range(dialog.ui.formatList.count()) if dialog.ui.formatList.item(i).expClass == CSVExporter)
for index in range(3):
    dialog.ui.formatList.setCurrentRow(image_row)
    for parameter in dialog.currentExporter.parameters().children():
        parameter.destroyed.connect(lambda _obj=None,label=f'{index}:{parameter.name()}':record('destroyed',name=label),Qt.ConnectionType.DirectConnection)
    del parameter
    dialog.ui.formatList.setCurrentRow(csv_row)
    app.sendPostedEvents(None,QEvent.Type.DeferredDelete)
    with ThreadPoolExecutor(max_workers=1) as pool:
        def collect():
            record('worker-before')
            record('worker-after',collected=gc.collect())
        pool.submit(collect).result()
plot.close()
plot.deleteLater()
app.sendPostedEvents(None,QEvent.Type.DeferredDelete)
record('complete')
